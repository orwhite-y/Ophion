CMOS MARKER LEGEND:
  0x01 = trigger fired (ept_hook)
  0x03 = heal enter
  0x04 = post-enter-cr3
  0x05 = pre-walk
  0x06 = post-walk
  0x07 = batch refresh CR3 switched
  0x08 = batch refresh walk start
  0x09 = batch refresh walk OK
  0x0A = batch refresh PTE loop done
  0x0B = match found
  0x0C = batch refresh path
  0x0D = batch refresh walk FAIL
  0x0F = triple fault
